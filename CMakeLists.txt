cmake_minimum_required(VERSION 3.22)
project(AURAVOXPro VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(AURAVOXPro COMPANY_NAME "Auravox Audio" BUNDLE_ID "com.auravox.audio.pro" PLUGIN_MANUFACTURER_CODE Auva PLUGIN_CODE Avp1 FORMATS VST3 PRODUCT_NAME "AURAVOX Pro" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics Filter Delay)
juce_generate_juce_header(AURAVOXPro)
target_sources(AURAVOXPro PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h)
target_compile_definitions(AURAVOXPro PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(AURAVOXPro PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
